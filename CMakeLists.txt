cmake_minimum_required(VERSION 3.24)
project(ndi_recorder LANGUAGES C CXX RC)
# Release date is explicit so rebuilding an existing release remains reproducible.
set(APP_RELEASE_DATE "20261002")
set(APP_VERSION "v${APP_RELEASE_DATE}")
string(SUBSTRING "${APP_RELEASE_DATE}" 0 4 APP_YEAR)
string(SUBSTRING "${APP_RELEASE_DATE}" 4 2 APP_MONTH)
string(SUBSTRING "${APP_RELEASE_DATE}" 6 2 APP_DAY)
# Windows VERSIONINFO components are 16-bit integers; do not use YYYYMMDD as one component.
string(REGEX REPLACE "^0" "" APP_MONTH "${APP_MONTH}")
string(REGEX REPLACE "^0" "" APP_DAY "${APP_DAY}")
set(APP_VERSION_NUMERIC "${APP_YEAR}.${APP_MONTH}.${APP_DAY}.0")
configure_file(src/version.h.in generated/version.h @ONLY)
configure_file(src/app.rc.in generated/app.rc @ONLY)
configure_file(src/app.manifest generated/app.manifest @ONLY)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Windows x64 required")
endif()
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 PROCESSINGNDILIB_STATIC)
add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/utf-8>" "$<$<COMPILE_LANGUAGE:CXX>:/EHsc>" "$<$<COMPILE_LANGUAGE:CXX>:/W3>")
set(IMGUI vendor/imgui)
set(FF vendor/ffmpeg)
add_library(sqlite STATIC vendor/sqlite/sqlite3.c)
target_compile_definitions(sqlite PRIVATE SQLITE_THREADSAFE=1)
add_library(core STATIC src/core.cpp)
target_include_directories(core PUBLIC src vendor vendor/sqlite)
target_link_libraries(core PUBLIC sqlite shell32 shlwapi psapi iphlpapi ws2_32 advapi32)
add_executable(ndi-recorder WIN32 "${CMAKE_CURRENT_BINARY_DIR}/generated/app.rc" src/main.cpp src/ui.cpp src/ndi.cpp src/media.cpp src/manager.cpp
 ${IMGUI}/imgui.cpp ${IMGUI}/imgui_draw.cpp ${IMGUI}/imgui_tables.cpp ${IMGUI}/imgui_widgets.cpp
 ${IMGUI}/backends/imgui_impl_win32.cpp ${IMGUI}/backends/imgui_impl_dx11.cpp)
target_link_options(ndi-recorder PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_BINARY_DIR}/generated/app.manifest")
target_include_directories(ndi-recorder PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated" ${IMGUI} ${IMGUI}/backends vendor/ndi/include ${FF}/include)
target_link_directories(ndi-recorder PRIVATE ${CMAKE_SOURCE_DIR}/${FF}/lib)
target_link_libraries(ndi-recorder PRIVATE core d3d11 dxgi d3dcompiler comdlg32 ole32 winmm avcodec avformat avutil avfilter swscale swresample)
file(GLOB FF_DLLS "${CMAKE_SOURCE_DIR}/${FF}/bin/*.dll")
add_custom_command(TARGET ndi-recorder POST_BUILD
 COMMAND ${CMAKE_COMMAND} -E copy_if_different ${FF_DLLS} "$<TARGET_FILE_DIR:ndi-recorder>"
 COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/vendor/ndi/bin/Processing.NDI.Lib.x64.dll" "$<TARGET_FILE_DIR:ndi-recorder>")
enable_testing()
add_executable(core-tests tests/core_tests.cpp)
target_link_libraries(core-tests PRIVATE core)
add_test(NAME core COMMAND core-tests)


add_executable(manager-tests tests/manager_tests.cpp src/manager.cpp src/ndi.cpp src/media.cpp)
target_include_directories(manager-tests PRIVATE vendor/ndi/include ${FF}/include)
target_link_directories(manager-tests PRIVATE ${CMAKE_SOURCE_DIR}/${FF}/lib)
target_link_libraries(manager-tests PRIVATE core d3d11 winmm avcodec avformat avutil avfilter swscale swresample)
add_dependencies(manager-tests ndi-recorder)
add_test(NAME manager COMMAND manager-tests)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
 add_test(NAME sync-checker COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_sync.py)
endif()

# The install manifest is the single source of truth for portable releases.
install(TARGETS ndi-recorder RUNTIME DESTINATION .)
install(FILES ${FF_DLLS} "${CMAKE_SOURCE_DIR}/vendor/ndi/bin/Processing.NDI.Lib.x64.dll" DESTINATION .)
install(FILES README.md README_EN.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION .)
install(FILES tools/Start-24h-Debug.cmd tools/Stop-24h-Debug.cmd tools/Inspect-Recording.cmd DESTINATION .)
install(FILES tools/longrun-debug.ps1 tools/inspect-recording.ps1 DESTINATION tools)
install(FILES "${CMAKE_SOURCE_DIR}/${FF}/bin/ffmpeg.exe" "${CMAKE_SOURCE_DIR}/${FF}/bin/ffprobe.exe" DESTINATION .)
install(DIRECTORY docs DESTINATION .)
install(FILES vendor/ffmpeg/LICENSE.txt DESTINATION licenses RENAME FFmpeg-LICENSE.txt)
install(FILES vendor/imgui/LICENSE.txt DESTINATION licenses RENAME ImGui-LICENSE.txt)
install(FILES tools/dependencies.json DESTINATION licenses)
install(FILES vendor/JSON-LICENSE.txt vendor/NDI-SDK-License.pdf DESTINATION licenses OPTIONAL)
set(CPACK_GENERATOR "ZIP")
set(CPACK_PACKAGE_NAME "ndi-recorder")
set(CPACK_PACKAGE_VERSION "${APP_VERSION}")
set(CPACK_PACKAGE_FILE_NAME "ndi-recorder-${APP_VERSION}-windows-x64")
set(CPACK_PACKAGE_DIRECTORY "${CMAKE_SOURCE_DIR}/dist")
set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY ON)
include(CPack)
